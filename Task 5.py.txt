def calculate_payment(quantity, fat, snf):
    """Calculate milk rate and payment for one delivery."""

    quantity = Decimal(quantity)
    fat = Decimal(fat)
    snf = Decimal(snf)

    validate_entry("TEMP", quantity, fat, snf)

    rate = calculate_rate(fat, snf)
    payment = round_money(quantity * rate)

    return rate, payment
