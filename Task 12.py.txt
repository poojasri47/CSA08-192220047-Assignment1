SEASONAL_BASE_RATE = Decimal("32.00")
SEASONAL_FAT_RATE_CHANGE = Decimal("0.45")

BONUS_THRESHOLD = Decimal("500.00")
BONUS_PER_LITRE = Decimal("0.50")


def calculate_seasonal_rate(fat, snf):
    """
    Calculate the April-June seasonal rate.

    Seasonal policy:
    - Base rate = ₹32.00 per litre
    - Each 0.1% fat step changes the rate by ₹0.45
    - SNF adjustment remains the original ₹0.30 per 0.1% step
    """

    fat = Decimal(fat)
    snf = Decimal(snf)

    validate_entry("TEMP", Decimal("1.00"), fat, snf)

    fat_steps = nearest_step(
        fat,
        BASE_FAT,
        FAT_STEP
    )

    snf_steps = nearest_step(
        snf,
        BASE_SNF,
        SNF_STEP
    )

    rate = (
        SEASONAL_BASE_RATE
        + Decimal(fat_steps) * SEASONAL_FAT_RATE_CHANGE
        + Decimal(snf_steps) * SNF_RATE_CHANGE
    )

    return round_money(rate)


def calculate_ten_day_seasonal_payment(entries):
    """
    Calculate ten-day seasonal payment and bonus.

    The bonus applies to all litres when the farmer's total
    ten-day supply is greater than 500 litres.
    """

    total_litres = Decimal("0.00")
    total_milk_payment = Decimal("0.00")

    for entry in entries:
        farmer_id = entry["farmer_id"]
        quantity = Decimal(entry["quantity"])
        fat = Decimal(entry["fat"])
        snf = Decimal(entry["snf"])

        validate_entry(
            farmer_id,
            quantity,
            fat,
            snf
        )

        rate = calculate_seasonal_rate(fat, snf)
        payment = round_money(quantity * rate)

        total_litres += quantity
        total_milk_payment += payment

    total_milk_payment = round_money(total_milk_payment)

    if total_litres > BONUS_THRESHOLD:
        bonus = round_money(
            total_litres * BONUS_PER_LITRE
        )
    else:
        bonus = Decimal("0.00")

    final_payment = round_money(
        total_milk_payment + bonus
    )

    return {
        "total_litres": total_litres,
        "milk_payment": total_milk_payment,
        "bonus": bonus,
        "final_payment": final_payment
    }
